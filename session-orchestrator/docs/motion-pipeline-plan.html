The wait utility

// motion.js. One function for all waits. One limit: --t-limit.
orchMotion.wait(target, "settle")              // all finite animations of the subtree
orchMotion.wait(target, "property", "clip-path") // one transition or animation name; none running: ends at once
orchMotion.wait(null,   "time", "--t-peek-show") // the time of a token
orchMotion.wait(target, "event", "frames:2")    // a DOM event, or n display frames
// Result: { by: "end" | "limit" | "hidden" | "missing", ms }
// by "limit" or "missing": C# writes a trace line with the step name.

The C# call: one command on the Umpire pipeline

// Features/Motion/Slices/Wait. The JS name comes from the type: MotionWait → orchMotion.wait.
public sealed record MotionWait(object Target, WaitMode Mode, string? Arg = null) : JsCommand<WaitResult>;
public sealed class MotionWaitHandler(IJSRuntime js, JsCalls calls) : JsCommandHandler<MotionWait, WaitResult>(js, calls);

// A call site. The handler catches JS errors and returns an Error; nothing is swallowed.
var r = await Commands.SendAsync(new MotionWait(stage, WaitMode.Property, "clip-path"));
